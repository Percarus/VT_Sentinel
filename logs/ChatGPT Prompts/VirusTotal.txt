import vt
import os
import time
import hashlib
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
import threading

API_KEY = "YOUR_API_KEY_HERE"
WATCHED_FOLDER = os.path.expanduser("~/Downloads")
HASH_RECORD_FILE = "scanned_files.db"
EXTENSIONS_TO_WATCH = (".exe", ".zip")

client = vt.Client(API_KEY)

# Load existing hashes
if os.path.exists(HASH_RECORD_FILE):
    with open(HASH_RECORD_FILE, "r") as f:
        scanned_hashes = set(line.strip() for line in f)
else:
    scanned_hashes = set()

def file_hash(path):
    sha256 = hashlib.sha256()
    with open(path, 'rb') as f:
        while chunk := f.read(8192):
            sha256.update(chunk)
    return sha256.hexdigest()

def scan_file(path):
    try:
        hash_ = file_hash(path)
        if hash_ in scanned_hashes:
            print(f"Already scanned: {path}")
            return
        with open(path, "rb") as f:
            print(f"Uploading: {path}")
            analysis = client.scan_file(f)

        # Wait for results
        while True:
            analysis = client.get_object("/analyses/{}", analysis.id)
            if analysis.status == "completed":
                break
            time.sleep(5)

        file_obj = client.get_object("/files/{}", analysis.meta["file_info"]["sha256"])
        stats = file_obj.last_analysis_stats
        print(f"Scan complete for {path}")
        print(stats)

        # Store hash to avoid rescan
        with open(HASH_RECORD_FILE, "a") as f:
            f.write(hash_ + "\n")
        scanned_hashes.add(hash_)
    except Exception as e:
        print(f"Error scanning {path}: {e}")

class FileHandler(FileSystemEventHandler):
    def on_created(self, event):
        if not event.is_directory and event.src_path.lower().endswith(EXTENSIONS_TO_WATCH):
            threading.Thread(target=scan_file, args=(event.src_path,)).start()

def run():
    print(f"Watching folder: {WATCHED_FOLDER}")
    observer = Observer()
    observer.schedule(FileHandler(), WATCHED_FOLDER, recursive=True)
    observer.start()
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        print("Stopping watcher...")
        observer.stop()
    observer.join()
    client.close()

if __name__ == "__main__":
    run()

                    # open browser or download file
                    import webbrowser

        print("Update check failed:", e)
